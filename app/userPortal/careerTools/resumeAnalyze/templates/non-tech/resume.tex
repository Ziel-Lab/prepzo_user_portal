{# Safe LaTeX escaping macro - use as {{ latex_escape(value) }} #}
{% macro latex_escape(text) -%}
{{ (text | default(''))
    | replace('\\', '\\textbackslash{}')
    | replace('#', '\\#')
    | replace('%', '\\%')
    | replace('$', '\\$')
    | replace('&', '\\&')
    | replace('_', '\\_')
    | replace('{', '\\{')
    | replace('}', '\\}')
    | replace('~', '\\textasciitilde{}')
    | replace('^', '\\textasciicircum{}')
}}
{%- endmacro %}
